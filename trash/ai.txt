```rust
use anyhow::Result;
use clap::Parser;
use tracing_subscriber::{layer::SubscriberExt, util::SubscriberInitExt};

mod commands;
mod status;

use commands::{Cli, Commands};
use status::{OutputFormat, StatusReporter};

#[tokio::main]
async fn main() -> Result<()> {
    // Initialize tracing
    tracing_subscriber::registry()
        .with(tracing_subscriber::EnvFilter::new(
            std::env::var("DX_LOG").unwrap_or_else(|_| "dx=info".into()),
        ))
        .with(tracing_subscriber::fmt::layer())
        .init();

    let cli = Cli::parse();
    let status = StatusReporter::new(cli.output_format());

    let result = match cli.command {
        Commands::Init(args) => commands::init::run(args, &status).await,
        Commands::Run(args) => commands::run::run(args, &status).await,
        Commands::Daemon(args) => commands::daemon::run(args, &status).await,
        Commands::Serializer(args) => commands::serializer::run(args, &status).await,
        Commands::Workspace(args) => commands::workspace::run(args, &status).await,
        Commands::Plugin(args) => commands::plugin::run(args, &status).await,
        Commands::Agent(args) => commands::agent::run(args, &status).await,
        Commands::Status(args) => commands::status::run(args, &status).await,
    };

    if let Err(e) = result {
        status.error(&e.to_string());
        std::process::exit(1);
    }

    Ok(())
}
```

```rust
use anyhow::Result;
use clap::{Args, Subcommand};

use crate::status::StatusReporter;

#[derive(Args)]
pub struct DaemonArgs {
    #[command(subcommand)]
    pub command: DaemonCommands,
}

#[derive(Subcommand)]
pub enum DaemonCommands {
    /// Start the daemon (runs 24/7, minimal CPU when idle)
    Start {
        /// Run in foreground (don't daemonize)
        #[arg(short, long)]
        foreground: bool,

        /// PID file path
        #[arg(long, default_value = "dx.pid")]
        pid_file: String,
    },

    /// Stop the daemon
    Stop,

    /// Restart the daemon
    Restart,

    /// Show daemon status
    Status,

    /// Send a command to the running daemon
    Send {
        /// Command to send
        command: String,
    },
}

pub async fn run(args: DaemonArgs, status: &StatusReporter) -> Result<()> {
    match args.command {
        DaemonCommands::Start { foreground, pid_file } => {
            if foreground {
                status.info("Starting daemon in foreground...");
                status.info("Daemon uses minimal CPU when idle (Rust async/await)");
                daemon::Daemon::run_foreground().await?;
            } else {
                status.info("Starting daemon in background...");
                daemon::Daemon::start(&pid_file)?;
                status.success("Daemon started");
            }
        }

        DaemonCommands::Stop => {
            status.info("Stopping daemon...");
            daemon::Daemon::stop()?;
            status.success("Daemon stopped");
        }

        DaemonCommands::Restart => {
            status.info("Restarting daemon...");
            daemon::Daemon::stop()?;
            daemon::Daemon::start("dx.pid")?;
            status.success("Daemon restarted");
        }

        DaemonCommands::Status => {
            let running = daemon::Daemon::is_running();
            if running {
                status.success("Daemon is running");
                // Show stats
                if let Some(stats) = daemon::Daemon::get_stats() {
                    status.info(&format!("  Uptime: {}", stats.uptime));
                    status.info(&format!("  CPU: {:.2}%", stats.cpu_percent));
                    status.info(&format!("  Memory: {} MB", stats.memory_mb));
                    status.info(&format!("  Workspaces: {}", stats.workspaces));
                    status.info(&format!("  Plugins: {}", stats.plugins_loaded));
                }
            } else {
                status.warning("Daemon is not running");
            }
        }

        DaemonCommands::Send { command } => {
            status.info(&format!("Sending command: {}", command));
            let response = daemon::Daemon::send_command(&command)?;
            println!("{}", response);
        }
    }

    Ok(())
}
```

```rust
use anyhow::Result;
use clap::{Args, Subcommand, ValueEnum};
use std::path::PathBuf;

use crate::status::StatusReporter;

#[derive(Args)]
pub struct PluginArgs {

    pub command: PluginCommands,
}

#[derive(Clone, ValueEnum)]
pub enum PluginType {

    /// LLM providers (Python → WASM)
    Provider,
    /// Performance tools (Rust/Go/C/Zig → WASM)
    Tool,
}

#[derive(Subcommand)]
pub enum PluginCommands {
    /// Add a new plugin
    Add {
        /// Plugin type
        #[arg(short = 't', long, value_enum)]
        r#type: PluginType,

        /// Plugin name
        name: String,

        /// Source path or URL
        #[arg(short, long)]
        source: Option<String>,
    },

    /// List installed plugins
    List {
        /// Filter by type
        #[arg(short = 't', long, value_enum)]
        r#type: Option<PluginType>,
    },

    /// Build plugin to WASM
    Build {
        /// Plugin path
        path: PathBuf,
    },

    /// Remove a plugin
    Remove {
        /// Plugin name
        name: String,
    },

    /// Show plugin info
    Info {
        /// Plugin name
        name: String,
    },
}

pub async fn run(args: PluginArgs, status: &StatusReporter) -> Result<()> {
    match args.command {
        PluginCommands::Add { r#type, name, source } => {
            match r#type {
                PluginType::Integration => {
                    status.info(&format!("Adding integration: {}", name));
                    status.info("→ Language: Node.js → WASM (Javy)");
                    status.info("→ Use case: WhatsApp, Telegram, Discord, X, Slack");
                }
                PluginType::Provider => {
                    status.info(&format!("Adding provider: {}", name));
                    status.info("→ Language: Python → WASM (componentize-py)");
                    status.info("→ Use case: OpenAI, Anthropic, Ollama, Gemini, HuggingFace");
                }

                PluginType::Provider => PathBuf::from("plugins/providers").join(&name),
                PluginType::Tool => PathBuf::from("plugins/tools").join(&name),
            };

            std::fs::create_dir_all(&plugin_dir)?;

            match r#type {
                PluginType::Integration => create_integration_scaffold(&plugin_dir, &name)?,
                PluginType::Provider => create_provider_scaffold(&plugin_dir, &name)?,
                PluginType::Tool => create_tool_scaffold(&plugin_dir, &name)?,

#[derive(Debug)]
enum Language {
    NodeJs,
    Python,
    Rust,
    Go,
    C,
    Cpp,
    Zig,
}

fn detect_plugin_type(path: &PathBuf) -> Result<PluginType> {
    let path_str = path.to_string_lossy();
    if path_str.contains("integrations") {
        Ok(PluginType::Integration)
    } else if path_str.contains("providers") {
        Ok(PluginType::Provider)
    } else if path_str.contains("tools") {
        Ok(PluginType::Tool)
    } else {
        // Detect by files
        if path.join("package.json").exists() {
            Ok(PluginType::Integration)
        } else if path.join("pyproject.toml").exists() {
            Ok(PluginType::Provider)
        } else {
            Ok(PluginType::Tool)
        }
    }
}

fn detect_language(path: &PathBuf) -> Result<Language> {
    if path.join("package.json").exists() {
        Ok(Language::NodeJs)
    } else if path.join("pyproject.toml").exists() || path.join("setup.py").exists() {
        Ok(Language::Python)
    } else if path.join("Cargo.toml").exists() {
        Ok(Language::Rust)
    } else if path.join("go.mod").exists() {
        Ok(Language::Go)
    } else if path.join("build.zig").exists() {
        Ok(Language::Zig)

fn to_pascal_case(s: &str) -> String {
    s.split(|c: char| c == '-' || c == '_')

                Some(c) => c.to_uppercase().collect::<String>() + chars.as_str(),
                None => String::new(),
            }
        })
        .collect()
}
```

```rust
//! DX Daemon - Runs 24/7 with minimal CPU usage when idle
//!
//! Design principles:
//! - Uses Rust async/await (Tokio) for efficient event-driven execution
//! - When idle, consumes near-zero CPU (just waiting on epoll/kqueue)
//! - Wakes up only on:
//!   - IPC commands (Unix socket / named pipe)
//!   - Signal handlers (SIGTERM, SIGINT, SIGHUP)
//!   - Timer events (scheduled tasks)
//!   - Plugin messages

use std::path::PathBuf;
use std::sync::Arc;

use futures::StreamExt;
use signal_hook::consts::{SIGINT, SIGTERM, SIGHUP};
use signal_hook_tokio::Signals;
use thiserror::Error;
use tokio::sync::{mpsc, RwLock, broadcast};
use tokio::net::UnixListener;
use tokio::io::{AsyncBufReadExt, AsyncWriteExt, BufReader};
use tracing::{info, warn, error};

pub mod ipc;
pub mod scheduler;

#[derive(Error, Debug)]
pub enum DaemonError {
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),

    #[error("Already running")]
    AlreadyRunning,

    #[error("Not running")]
    NotRunning,

    #[error("IPC error: {0}")]
    Ipc(String),
}

pub type Result<T> = std::result::Result<T, DaemonError>;

/// Daemon statistics
#[derive(Debug, Clone, Default)]
pub struct DaemonStats {
    pub start_time: Option<std::time::Instant>,
    pub cpu_percent: f32,
    pub memory_mb: u64,
    pub workspaces: usize,
    pub plugins_loaded: usize,
    pub commands_processed: u64,
}

impl DaemonStats {
    pub fn uptime(&self) -> String {
        match self.start_time {
            Some(start) => {
                let duration = start.elapsed();
                let secs = duration.as_secs();
                let hours = secs / 3600;
                let mins = (secs % 3600) / 60;
                let secs = secs % 60;
                format!("{}h {}m {}s", hours, mins, secs)
            }
            None => "0s".to_string(),
        }
    }
}

/// Internal daemon state
struct DaemonState {
    stats: DaemonStats,
    workspace_manager: workspace::WorkspaceManager,
    plugin_host: plugins::PluginHost,
    shutdown: broadcast::Sender<()>,
}

/// The main daemon
pub struct Daemon {
    socket_path: PathBuf,
    pid_file: PathBuf,
    state: Arc<RwLock<DaemonState>>,
}

impl Daemon {
    /// Create a new daemon instance
    pub fn new() -> Result<Self> {
        let socket_path = PathBuf::from("/tmp/dx.sock");
        let pid_file = PathBuf::from("dx.pid");

        let (shutdown_tx, _) = broadcast::channel(1);

        let state = DaemonState {
            stats: DaemonStats {
                start_time: Some(std::time::Instant::now()),
                ..Default::default()
            },
            workspace_manager: workspace::WorkspaceManager::new(workspace::ResourceLimits::default()),
            plugin_host: plugins::PluginHost::new().map_err(|e| DaemonError::Io(std::io::Error::other(e.to_string())))?,
            shutdown: shutdown_tx,
        };

        Ok(Self {
            socket_path,
            pid_file,
            state: Arc::new(RwLock::new(state)),
        })
    }

    /// Run the daemon in foreground (blocking)
    pub async fn run_foreground() -> Result<()> {
        let daemon = Self::new()?;
        daemon.run().await
    }

    /// Start daemon in background
    pub fn start(pid_file: &str) -> Result<()> {
        // Fork and daemonize (Unix)
        #[cfg(unix)]
        {
            use std::process::Command;
            
            let exe = std::env::current_exe()?;
            let child = Command::new(exe)
                .args(["daemon", "start", "--foreground"])
                .spawn()?;
            
            std::fs::write(pid_file, child.id().to_string())?;
        }

        #[cfg(windows)]
        {
            // Windows service or background process
            todo!("Windows daemon support")
        }

        Ok(())
    }

    /// Stop the daemon
    pub fn stop() -> Result<()> {
        // Send shutdown command via IPC
        let response = Self::send_command("shutdown")?;
        if response.contains("ok") {
            Ok(())
        } else {
            Err(DaemonError::Ipc(response))
        }
    }

    /// Check if daemon is running
    pub fn is_running() -> bool {
        // Check if socket exists and responds
        Self::send_command("ping").is_ok()
    }

    /// Get daemon statistics
    pub fn get_stats() -> Option<DaemonStats> {
        let response = Self::send_command("stats").ok()?;
        serde_json::from_str(&response).ok()
    }

    /// Send command to running daemon
    pub fn send_command(command: &str) -> Result<String> {
        use std::io::{Read, Write};
        use std::os::unix::net::UnixStream;

        let mut stream = UnixStream::connect("/tmp/dx.sock")
            .map_err(|_| DaemonError::NotRunning)?;
        
        stream.write_all(command.as_bytes())?;
        stream.write_all(b"\n")?;
        stream.flush()?;

        let mut response = String::new();
        stream.read_to_string(&mut response)?;
        
        Ok(response)
    }

    /// Main daemon loop
    async fn run(&self) -> Result<()> {
        info!("Starting DX daemon...");

        // Remove old socket if exists
        let _ = std::fs::remove_file(&self.socket_path);

        // Create Unix socket listener
        let listener = UnixListener::bind(&self.socket_path)?;
        info!("Listening on {:?}", self.socket_path);

        // Setup signal handlers
        let mut signals = Signals::new([SIGINT, SIGTERM, SIGHUP])?;

        // Get shutdown receiver
        let mut shutdown_rx = {
            let state = self.state.read().await;
            state.shutdown.subscribe()
        };

        info!("Daemon running. Waiting for commands (minimal CPU when idle)...");

        loop {
            tokio::select! {
                // Handle incoming IPC connections
                // This is where we wait efficiently - no CPU usage while waiting
                result = listener.accept() => {
                    match result {
                        Ok((stream, _addr)) => {
                            let state = Arc::clone(&self.state);
                            tokio::spawn(async move {
                                if let Err(e) = Self::handle_connection(stream, state).await {
                                    error!("Connection error: {}", e);
                                }
                            });
                        }
                        Err(e) => {
                            error!("Accept error: {}", e);
                        }
                    }
                }

                // Handle OS signals
                Some(signal) = signals.next() => {
                    match signal {
                        SIGINT | SIGTERM => {
                            info!("Received shutdown signal");
                            break;
                        }
                        SIGHUP => {
                            info!("Received SIGHUP, reloading configuration...");
                            // TODO: Reload config
                        }
                        _ => {}
                    }
                }

                // Handle shutdown broadcast
                _ = shutdown_rx.recv() => {
                    info!("Received shutdown command");
                    break;
                }
            }
        }

        info!("Daemon shutting down...");
        
        // Cleanup
        let _ = std::fs::remove_file(&self.socket_path);
        let _ = std::fs::remove_file(&self.pid_file);

        Ok(())
    }

    /// Handle a single IPC connection
    async fn handle_connection(
        stream: tokio::net::UnixStream,
        state: Arc<RwLock<DaemonState>>,
    ) -> Result<()> {
        let (reader, mut writer) = stream.into_split();
        let mut reader = BufReader::new(reader);
        let mut line = String::new();

        // Read command
        reader.read_line(&mut line).await?;
        let command = line.trim();

        // Increment command counter
        {
            let mut state = state.write().await;
            state.stats.commands_processed += 1;
        }

        // Process command
        let response = match command {
            "ping" => "pong".to_string(),
            
            "shutdown" => {
                let state = state.read().await;
                let _ = state.shutdown.send(());
                r#"{"status":"ok","msg":"shutting down"}"#.to_string()
            }
            
            "stats" => {
                let state = state.read().await;
                serde_json::to_string(&DaemonStatsResponse {
                    uptime: state.stats.uptime(),
                    cpu_percent: state.stats.cpu_percent,
                    memory_mb: state.stats.memory_mb,
                    workspaces: state.stats.workspaces,
                    plugins_loaded: state.stats.plugins_loaded,
                    commands_processed: state.stats.commands_processed,
                }).unwrap_or_else(|_| "{}".to_string())
            }
            
            "status" => {
                r#"{"status":"running"}"#.to_string()
            }
            
            cmd if cmd.starts_with("workspace ") => {
                let args = &cmd[10..];
                // Handle workspace commands
                format!(r#"{{"status":"ok","workspace":"{}"}}"#, args)
            }
            
            cmd if cmd.starts_with("plugin ") => {
                let args = &cmd[7..];
                // Handle plugin commands
                format!(r#"{{"status":"ok","plugin":"{}"}}"#, args)
            }
            
            cmd if cmd.starts_with("exec ") => {
                let cmd_to_exec = &cmd[5..];
                // Execute in sandbox
                let mut state = state.write().await;
                match state.workspace_manager.execute_in_primary(cmd_to_exec).await {
                    Ok(output) => format!(r#"{{"status":"ok","output":"{}"}}"#, output.replace('"', "\\\"")),
                    Err(e) => format!(r#"{{"status":"error","msg":"{}"}}"#, e),
                }
            }
            
            _ => {
                format!(r#"{{"status":"error","msg":"unknown command: {}"}}"#, command)
            }
        };

        // Send response
        writer.write_all(response.as_bytes()).await?;
        writer.write_all(b"\n").await?;
        writer.flush().await?;

        Ok(())
    }
}

    uptime: String,
    cpu_percent: f32,
    memory_mb: u64,
    workspaces: usize,
    plugins_loaded: usize,
    commands_processed: u64,
}
```

```rust
//! Task scheduler for the daemon

use std::collections::BinaryHeap;
use std::cmp::Ordering;
use std::time::{Duration, Instant};

use tokio::sync::mpsc;

/// A scheduled task
#[derive(Debug)]
pub struct ScheduledTask {
    pub id: u64,
    pub run_at: Instant,
    pub command: String,
    pub repeat: Option<Duration>,
}

impl Eq for ScheduledTask {}

impl PartialEq for ScheduledTask {
    fn eq(&self, other: &Self) -> bool {
        self.id == other.id
    }
}

impl Ord for ScheduledTask {
    fn cmp(&self, other: &Self) -> Ordering {
        // Reverse ordering for min-heap (earliest first)
        other.run_at.cmp(&self.run_at)
    }
}

impl PartialOrd for ScheduledTask {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

/// Task scheduler
pub struct Scheduler {
    tasks: BinaryHeap<ScheduledTask>,
    next_id: u64,
    tx: mpsc::Sender<ScheduledTask>,
}

impl Scheduler {
    pub fn new() -> (Self, mpsc::Receiver<ScheduledTask>) {
        let (tx, rx) = mpsc::channel(100);
        (
            Self {
                tasks: BinaryHeap::new(),
                next_id: 0,
                tx,
            },
            rx,
        )
    }

    /// Schedule a task to run at a specific time
    pub fn schedule_at(&mut self, run_at: Instant, command: String) -> u64 {
        let id = self.next_id;
        self.next_id += 1;

        self.tasks.push(ScheduledTask {
            id,
            run_at,
            command,
            repeat: None,
        });

        id
    }

    /// Schedule a task to run after a delay
    pub fn schedule_after(&mut self, delay: Duration, command: String) -> u64 {
        self.schedule_at(Instant::now() + delay, command)
    }

    /// Schedule a repeating task
    pub fn schedule_repeat(&mut self, interval: Duration, command: String) -> u64 {
        let id = self.next_id;
        self.next_id += 1;

        self.tasks.push(ScheduledTask {
            id,
            run_at: Instant::now() + interval,
            command,
            repeat: Some(interval),
        });

        id
    }

    /// Get the next task if it's ready to run
    pub fn poll(&mut self) -> Option<ScheduledTask> {
        if let Some(task) = self.tasks.peek() {
            if task.run_at <= Instant::now() {
                let mut task = self.tasks.pop().unwrap();
                
                // If repeating, reschedule
                if let Some(interval) = task.repeat {
                    self.tasks.push(ScheduledTask {
                        id: self.next_id,
                        run_at: Instant::now() + interval,
                        command: task.command.clone(),
                        repeat: Some(interval),
                    });
                    self.next_id += 1;
                }

                return Some(task);
            }
        }
        None
    }

    /// Get time until next task (for sleep duration)
    pub fn time_until_next(&self) -> Option<Duration> {
        self.tasks.peek().map(|task| {
            let now = Instant::now();
            if task.run_at > now {
                task.run_at - now
            } else {
                Duration::ZERO
            }
        })
    }
}

impl Default for Scheduler {
    fn default() -> Self {
        Self::new().0
    }
}
```

```rust
use crate::{PluginError, Result};

/// Compile Node.js integration to WASM

    let compiler = NodeJsCompiler::new();
    
    if !compiler.is_available().await {
        return Err(PluginError::Compilation(
            "Javy not installed. Run: cargo install javy-cli".into()
        ));
    }

/// Node.js to WASM compiler using Javy
pub struct NodeJsCompiler {
    javy_command: String,
    esbuild_command: String,
}

impl NodeJsCompiler {
    pub fn new() -> Self {
        Self {
            javy_command: "javy".to_string(),
            esbuild_command: "npx".to_string(),
        }
    }

    /// Check if required tools are available
    pub async fn is_available(&self) -> bool {
        let javy_ok = tokio::process::Command::new(&self.javy_command)
            .arg("--version")
            .output()
            .await
            .map(|o| o.status.success())
            .unwrap_or(false);

        let npm_ok = tokio::process::Command::new("npm")
            .arg("--version")
            .output()
            .await
            .map(|o| o.status.success())
            .unwrap_or(false);

        javy_ok && npm_ok
    }

    /// Compile Node.js/TypeScript to WASM
    pub async fn compile(&self, source_dir: &Path) -> Result<Vec<u8>> {
        // 1. Install dependencies if needed
        if source_dir.join("package.json").exists() && !source_dir.join("node_modules").exists() {
            tracing::info!("Installing npm dependencies...");
            let status = tokio::process::Command::new("npm")
                .args(["install"])
                .current_dir(source_dir)
                .status()
                .await?;

            if !status.success() {
                return Err(PluginError::Compilation("npm install failed".into()));
            }
        }

        // 2. Bundle with esbuild
        tracing::info!("Bundling with esbuild...");
        std::fs::create_dir_all(source_dir.join("dist"))?;

        let entry_point = if source_dir.join("src/index.ts").exists() {
            "src/index.ts"
        } else if source_dir.join("src/index.js").exists() {
            "src/index.js"
        } else {
            return Err(PluginError::Compilation("No entry point found (src/index.ts or src/index.js)".into()));
        };

        let status = tokio::process::Command::new(&self.esbuild_command)
            .args([
                "esbuild",
                entry_point,
                "--bundle",
                "--outfile=dist/bundle.js",
                "--format=esm",
                "--platform=neutral",
                "--target=es2020",
            ])
            .current_dir(source_dir)
            .status()
            .await?;

        if !status.success() {
            return Err(PluginError::Compilation("esbuild bundling failed".into()));
        }

        // 3. Compile with Javy
        tracing::info!("Compiling to WASM with Javy...");
        let status = tokio::process::Command::new(&self.javy_command)
            .args(["compile", "dist/bundle.js", "-o", "plugin.wasm"])
            .current_dir(source_dir)
            .status()
            .await?;

        if !status.success() {
            return Err(PluginError::Compilation("Javy compilation failed".into()));
        }

        // 4. Read WASM bytes
        let wasm_path = source_dir.join("plugin.wasm");
        tokio::fs::read(&wasm_path).await.map_err(Into::into)
    }
}

impl Default for NodeJsCompiler {
    fn default() -> Self {
        Self::new()
    }
}
```

```rust
pub struct RustCompiler;

    pub async fn compile(source_dir: &Path) -> Result<Vec<u8>> {

            })
            .ok_or_else(|| PluginError::Compilation("WASM output not found".into()))?;
```

```rust
//! MCP (Model Context Protocol) Tool for DX
//! Compiled to WASM - runs with native speed in sandbox

wit_bindgen::generate!({
    world: "dx-tool",
});

use exports::dx::tools::mcp::Guest;

struct McpTool;

impl Guest for McpTool {
    /// Handle MCP JSON-RPC request
    fn handle_request(request: String) -> String {
        let request: serde_json::Value = match serde_json::from_str(&request) {
            Ok(v) => v,
            Err(e) => return error_response(-32700, &format!("Parse error: {}", e)),
        };

        let method = request.get("method").and_then(|m| m.as_str()).unwrap_or("");
        let id = request.get("id").cloned().unwrap_or(serde_json::Value::Null);

        let result = match method {
            "initialize" => handle_initialize(&request),
            "tools/list" => handle_list_tools(),
            "tools/call" => handle_call_tool(&request),
            "resources/list" => handle_list_resources(),
            "resources/read" => handle_read_resource(&request),
            "ping" => Ok(serde_json::json!({})),
            _ => Err((-32601, format!("Method not found: {}", method))),
        };

        match result {
            Ok(result) => serde_json::json!({
                "jsonrpc": "2.0",
                "id": id,
                "result": result
            }).to_string(),
            Err((code, message)) => error_response_with_id(id, code, &message),
        }
    }

    /// Get MCP server info
    fn server_info() -> String {
        serde_json::json!({
            "name": "dx-mcp",
            "version": "0.1.0",
            "protocolVersion": "2024-11-05"
        }).to_string()
    }
}

fn handle_initialize(_request: &serde_json::Value) -> Result<serde_json::Value, (i32, String)> {
    Ok(serde_json::json!({
        "protocolVersion": "2024-11-05",
        "capabilities": {
            "tools": {},
            "resources": {}
        },
        "serverInfo": {
            "name": "dx-mcp",
            "version": "0.1.0"
        }
    }))
}

fn handle_list_tools() -> Result<serde_json::Value, (i32, String)> {
    Ok(serde_json::json!({
        "tools": [
            {
                "name": "execute_command",
                "description": "Execute a shell command in sandbox",
                "inputSchema": {
                    "type": "object",
                    "properties": {
                        "command": { "type": "string" }
                    },
                    "required": ["command"]
                }
            },
            {
                "name": "read_file",
                "description": "Read a file's contents",
                "inputSchema": {
                    "type": "object",
                    "properties": {
                        "path": { "type": "string" }
                    },
                    "required": ["path"]
                }
            }
        ]
    }))
}

fn handle_call_tool(request: &serde_json::Value) -> Result<serde_json::Value, (i32, String)> {
    let params = request.get("params").ok_or((-32602, "Missing params".into()))?;
    let name = params.get("name").and_then(|n| n.as_str()).ok_or((-32602, "Missing tool name".into()))?;
    let arguments = params.get("arguments").cloned().unwrap_or(serde_json::json!({}));

    match name {
        "execute_command" => {
            let command = arguments.get("command").and_then(|c| c.as_str()).unwrap_or("");
            // In WASM, we'd call back to the host for actual execution
            Ok(serde_json::json!({
                "content": [{
                    "type": "text",
                    "text": format!("Would execute: {}", command)
                }]
            }))
        }
        "read_file" => {
            let path = arguments.get("path").and_then(|p| p.as_str()).unwrap_or("");
            Ok(serde_json::json!({
                "content": [{
                    "type": "text",
                    "text": format!("Would read: {}", path)
                }]
            }))
        }
        _ => Err((-32602, format!("Unknown tool: {}", name)))
    }
}

fn handle_list_resources() -> Result<serde_json::Value, (i32, String)> {
    Ok(serde_json::json!({
        "resources": []
    }))
}

fn handle_read_resource(_request: &serde_json::Value) -> Result<serde_json::Value, (i32, String)> {
    Err((-32602, "Resource not found".into()))
}

fn error_response(code: i32, message: &str) -> String {
    serde_json::json!({
        "jsonrpc": "2.0",
        "id": null,
        "error": {
            "code": code,
            "message": message
        }
    }).to_string()
}

fn error_response_with_id(id: serde_json::Value, code: i32, message: &str) -> String {
    serde_json::json!({
        "jsonrpc": "2.0",
        "id": id,
        "error": {
            "code": code,
            "message": message
        }
    }).to_string()
}
```

```bash
cargo watch \
    --clear \
    --watch crates \
    --watch config.sr \
    -x 'build -p dx-cli -p dx-daemon' \
    -s './target/debug/dx status --ai'
```

### `scripts/build-plugins.sh`

```bash
#!/bin/bash
# Build all plugins to WASM

set -e

echo "🔨 Building DX plugins..."

# Build integrations (Node.js → WASM)
echo ""
echo "📱 Building Integrations (Node.js → WASM)..."
for dir in plugins/integrations/*/; do
    if [ -d "$dir" ]; then
        name=$(basename "$dir")
        echo "   → $name"
        ./target/release/dx plugin build "$dir" 2>/dev/null || echo "     (skipped - no source)"
    fi
done

# Build providers (Python → WASM)
echo ""
echo "🤖 Building Providers (Python → WASM)..."
for dir in plugins/providers/*/; do
    if [ -d "$dir" ]; then
        name=$(basename "$dir")
        echo "   → $name"
        ./target/release/dx plugin build "$dir" 2>/dev/null || echo "     (skipped - no source)"
    fi
done

# Build tools (Rust/Go/C/Zig → WASM)
echo ""
echo "🔧 Building Tools (Rust/Go/C/Zig → WASM)..."
for dir in plugins/tools/*/; do
    if [ -d "$dir" ]; then
        name=$(basename "$dir")
        echo "   → $name"
        ./target/release/dx plugin build "$dir" 2>/dev/null || echo "     (skipped - no source)"
    fi
done

echo ""
echo "✅ All plugins built!"
```

## Deployment

### `deploy/docker/Dockerfile`

```dockerfile
# Multi-stage build for minimal image

FROM rust:1.83-alpine AS builder

RUN apk add --no-cache musl-dev pkgconfig

WORKDIR /build
COPY . .

RUN cargo build --release -p dx-cli -p dx-daemon

# Runtime image (<50MB)
FROM alpine:3.20

RUN apk add --no-cache ca-certificates bubblewrap

RUN adduser -D -u 1000 dx

COPY --from=builder /build/target/release/dx /usr/local/bin/dx
COPY --from=builder /build/target/release/dx-daemon /usr/local/bin/dx-daemon

USER dx
WORKDIR /home/dx

# Daemon runs 24/7 with minimal CPU when idle
ENTRYPOINT ["dx-daemon"]
CMD ["start", "--foreground"]
```

## Usage

```bash
# Initialize project
dx init my-project --with-examples
cd my-project

# Start daemon (runs 24/7, minimal CPU when idle)
dx daemon start
dx daemon status

# Run interactive agent
dx run

# Manage plugins by category
dx plugin add --type integration discord   # Node.js → WASM
dx plugin add --type provider openai       # Python → WASM  
dx plugin add --type tool mcp              # Rust → WASM

dx plugin list                              # Show all
dx plugin list --type integration           # Show only integrations

# Build plugins
dx plugin build plugins/integrations/discord
dx plugin build plugins/providers/openai
dx plugin build plugins/tools/mcp

# Workspaces (parallel agents)
dx workspace spawn --name agent-1
dx workspace list

# Send command to daemon
dx daemon send "exec ls -la"
dx daemon send "status"

# Stop daemon
dx daemon stop
```
